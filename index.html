import cv2, turtle
img = cv2.imread('spiderman.png', 0)
_, thresh = cv2.threshold(img, 127, 255, THRESH_INV)
contours, _ = cv2.findContours(thresh, CHAIN_APPROX_NONE)
t = turtle.Turtle()
t.color('black'); t.begin_fill()
for pt in contour:
    t.goto(pt.x, pt.y)  # Drawing step-by-step
t.end_fill()  # Fill silhouette
